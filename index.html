    function handleDirectUpload(productId, inputElement) {

      const file = inputElement.files[0];

      if (!file) return;


      const reader = new FileReader();

      reader.onload = function(e) {

        userImages[productId] = e.target.result;

        renderProducts();

        triggerToast(`Updated photo for ${perfumes.find(x => x.id === productId).name}!`);

      };

      reader.readAsDataURL(file);

    }


    function addToCart(id) {

      cart[id] = (cart[id] || 0) + 1;

    function changeQty(id, delta) {

      if (!cart[id]) return;

      cart[id] += delta;

      if (cart[id] <= 0) {

        delete cart[id];

      }

      updateCartDisplay();

      renderCartList();

    }


    function updateCartDisplay() {

      let count = 0;

      let total = 0;


      for (let id in cart) {

        const p = perfumes.find(x => x.id === id);

        if (p) {

          count += cart[id];

          total += cart[id] * p.price;

        }

      }


      const bar = document.getElementById('floatingCart');

      document.getElementById('bagCount').innerText = `${count} item${count > 1 ? 's' : ''}`;

      document.getElementById('bagTotal').innerText = `₹${total}`;

      document.getElementById('modalTotal').innerText = `₹${total}`;


      if (count > 0) {

        bar.classList.remove('translate-y-full');

      } else {

        bar.classList.add('translate-y-full');

        closeCartModal();

      }

    }
